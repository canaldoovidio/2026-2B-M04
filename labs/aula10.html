Laboratório: o rastreador completo acorda uma vez por semana, publica a fila e volta a dormir

As camadas das Aulas 01 a 09 viram um único ciclo: lacre com debounce, estados do ciclo, fila de eventos, bateria, publicação MQTT, display e LED de status. Entre um despertar e outro o ESP32 fica em deep sleep. A lógica pura mora em rastreador.h, que você testa no computador, e a etapa Confrontar põe à prova os testes que a IA escreve.

Código (labs/aula10/sketch.ino) 

// Aula 10: ciclo completo do rastreador da Cerradinho, integrado e com deep sleep.
// A cada despertar: lê o lacre com debounce, lê a bateria, avança os dias fora,
// calcula o estado do ciclo, guarda os eventos na fila, publica a fila por MQTT
// quando há rede, mostra o resumo no display e no LED e volta a dormir.
// A lógica pura (estados, debounce, fila, bateria) está em rastreador.h.
// Tempo comprimido: cada despertar vale DIAS_POR_DESPERTAR dias do ciclo.
#include <WiFi.h>
#include <PubSubClient.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include "rastreador.h"

// Cadastro do ciclo CI-009 do conjunto fictício dados/ciclos.csv.
const char* ID_DISPOSITIVO = "DV-02";
const char* ID_EQUIPAMENTO = "EQ-02";
const char* NF_REMESSA = "NF-R-10063";
const int PRAZO_DIAS = 30;
const int LIMIAR_PERMANENCIA_DIAS = 150;   // referência inicial do TAPI, configurável

const int DIAS_POR_DESPERTAR = 7;          // um despertar por semana, como pede o TAPI
const uint64_t SEGUNDOS_DE_SONO = 5;       // 5 s de simulação no lugar de uma semana
#define USAR_DEEP_SLEEP 1                  // 0 troca o deep sleep por delay(), com o mesmo ciclo

const int PINO_LACRE = 27;     // chave deslizante: esquerda = lacre intacto (LOW)
const int PINO_RETORNO = 26;   // chave deslizante: direita = equipamento retornou
const int PINO_REDE = 25;      // chave deslizante: direita = sem conexão
const int PINO_BATERIA = 34;   // ADC1: o ADC2 é compartilhado com o Wi-Fi
const int PINO_LED = 4;        // LED de status
const uint32_t JANELA_DEBOUNCE_MS = 20;

const char* BROKER = "broker.hivemq.com";
const char* TOPICO = "m04-t24/rastreador/DV-02/eventos";

// Memória RTC: sobrevive ao deep sleep e se perde quando a energia é cortada.
RTC_DATA_ATTR bool cadastrado = false;
RTC_DATA_ATTR uint16_t diasFora = 0;
RTC_DATA_ATTR uint32_t despertares = 0;
RTC_DATA_ATTR uint8_t estadoAnterior = NO_PRAZO;
RTC_DATA_ATTR bool lacreRegistrado = false;
RTC_DATA_ATTR uint32_t publicados = 0;
RTC_DATA_ATTR FilaEventos fila;

WiFiClient clienteRede;
PubSubClient mqtt(clienteRede);
LiquidCrystal_I2C lcd(0x27, 16, 2);
unsigned long inicioMs = 0;

// Espera o contato ficar parado pela janela inteira, por até 500 ms.
int lerEstavel(int pino) {
  unsigned long t0 = millis();
  Debounce d = novoDebounce(digitalRead(pino), t0);
  while (millis() - t0 < 500) {
    atualizarDebounce(d, digitalRead(pino), millis(), JANELA_DEBOUNCE_MS);
    if (d.anterior == d.estavel && millis() - d.mudouEmMs >= JANELA_DEBOUNCE_MS) break;
  }
  return d.estavel;
}

bool conectar() {
  WiFi.begin("Wokwi-GUEST", "", 6);
  unsigned long t0 = millis();
  while (WiFi.status() != WL_CONNECTED && millis() - t0 < 10000) delay(100);
  if (WiFi.status() != WL_CONNECTED) return false;
  mqtt.setServer(BROKER, 1883);
  String id = String("m04-t24-") + ID_DISPOSITIVO + "-" + String((uint32_t)esp_random(), HEX);
  return mqtt.connect(id.c_str());
}

// Publica do mais antigo ao mais novo; o evento só sai da fila depois de publicado.
void publicarFila(bool lacreRompido) {
  Evento e;
  while (primeiroEvento(fila, e)) {
    char json[200];
    snprintf(json, sizeof(json),
             "{\"dispositivo\":\"%s\",\"equipamento\":\"%s\",\"nf\":\"%s\",\"dia\":%u,"
             "\"evento\":\"%s\",\"bateria\":%u,\"lacre\":\"%s\",\"perdidos\":%u}",
             ID_DISPOSITIVO, ID_EQUIPAMENTO, NF_REMESSA, e.dia, nomeDoEvento(e.tipo),
             e.bateriaPct, lacreRompido ? "rompido" : "intacto", fila.perdidos);
    if (!mqtt.publish(TOPICO, json)) break;
    Serial.println(json);
    descartarPrimeiro(fila);
    publicados++;
  }
}

void mostrar(EstadoCiclo estado, bool lacreRompido, int bateria) {
  const char* curto[] = { "NO PRAZO", "ATRASO", "PERMANENC.", "RETORNO" };
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.printf("D%u %s", diasFora, curto[estado]);
  lcd.setCursor(0, 1);
  lcd.printf("B%d%% F%u L:%s", bateria, fila.quantidade, lacreRompido ? "ROMP" : "OK");
  bool alerta = lacreRompido || estado == ATRASADO || estado == PERMANENCIA_LONGA;
  digitalWrite(PINO_LED, alerta ? HIGH : LOW);
}

void despertar() {
  despertares++;
  Serial.printf("\nDespertar %lu, causa %d\n", (unsigned long)despertares, (int)esp_sleep_get_wakeup_cause());
  int bateria = percentualBateria(analogReadMilliVolts(PINO_BATERIA) * 2);   // divisor 1:2
  bool lacreRompido = lerEstavel(PINO_LACRE) == HIGH;
  bool retornou = lerEstavel(PINO_RETORNO) == HIGH;
  bool semRede = lerEstavel(PINO_REDE) == HIGH;

  if (!cadastrado) {                                   // primeiro despertar: cadastro do ciclo
    limparFila(fila);
    cadastrado = true;
    diasFora = 0;
    estadoAnterior = NO_PRAZO;
    publicados = 0;
    Evento e = { 0, EV_CADASTRO, (uint8_t)bateria };
    guardarEvento(fila, e);
  } else {
    diasFora += DIAS_POR_DESPERTAR;
    Evento e = { diasFora, EV_LEITURA, (uint8_t)bateria };
    guardarEvento(fila, e);
  }

  if (lacreRompido && !lacreRegistrado) {
    Evento e = { diasFora, EV_LACRE, (uint8_t)bateria };
    guardarEvento(fila, e);
  }
  lacreRegistrado = lacreRompido;                      // lacre reaplicado volta a ser vigiado

  EstadoCiclo estado = estadoDoCiclo(diasFora, PRAZO_DIAS, LIMIAR_PERMANENCIA_DIAS, retornou);
  int alerta = eventoDaTransicao((EstadoCiclo)estadoAnterior, estado);
  if (alerta >= 0) {
    Evento e = { diasFora, (uint8_t)alerta, (uint8_t)bateria };
    guardarEvento(fila, e);
  }
  estadoAnterior = estado;

  if (!semRede && conectar()) publicarFila(lacreRompido);
  mostrar(estado, lacreRompido, bateria);
  Serial.printf("dia %u | %s | lacre %s | bateria %d%% | fila %u | publicados %lu | perdidos %u\n",
                diasFora, nomeDoEstado(estado), lacreRompido ? "rompido" : "intacto", bateria,
                fila.quantidade, (unsigned long)publicados, fila.perdidos);
}

void dormir() {
  Serial.printf("Acordado por %lu ms. Dormindo %lu s.\n", millis() - inicioMs, (unsigned long)SEGUNDOS_DE_SONO);
  mqtt.disconnect();
  WiFi.disconnect(true);
  WiFi.mode(WIFI_OFF);
  bool encerrado = estadoAnterior == RETORNADO;
  if (encerrado) {
    cadastrado = false;                                // dispositivo liberado para outro ciclo
    Serial.println("Ciclo encerrado. Para um novo cadastro, volte a chave de retorno e reinicie.");
  }
#if USAR_DEEP_SLEEP
  if (!encerrado) esp_sleep_enable_timer_wakeup(SEGUNDOS_DE_SONO * 1000000ULL);  // sem timer, só o reset acorda
  Serial.flush();
  esp_deep_sleep_start();
#else
  delay(SEGUNDOS_DE_SONO * 1000);
  while (encerrado) delay(1000);
#endif
}

void setup() {
  inicioMs = millis();
  Serial.begin(115200);
  pinMode(PINO_LACRE, INPUT_PULLUP);
  pinMode(PINO_RETORNO, INPUT_PULLUP);
  pinMode(PINO_REDE, INPUT_PULLUP);
  pinMode(PINO_LED, OUTPUT);
  lcd.init();
  lcd.backlight();
}

void loop() {
  inicioMs = millis();
  despertar();
  dormir();
}

Lógica pura (labs/aula10/rastreador.h) 

// rastreador.h: lógica pura do rastreador da Cerradinho (Aula 10).
// Nada aqui depende do Arduino: o mesmo arquivo compila no ESP32, incluído
// pelo sketch.ino, e no computador, incluído por testes/teste_rastreador.cpp.
// Quem lê pino, relógio ou rede é o sketch; aqui só entram números.
#pragma once
#include <stdint.h>

// ---------- Estados do ciclo ----------
enum EstadoCiclo : uint8_t { NO_PRAZO = 0, ATRASADO = 1, PERMANENCIA_LONGA = 2, RETORNADO = 3 };

// O TAPI separa atraso (passou do prazo previsto) de permanência prolongada
// (passou do limiar configurável, com referência inicial de 150 dias).
inline EstadoCiclo estadoDoCiclo(int diasFora, int prazoDias, int limiarDias, bool retornou) {
  if (retornou) return RETORNADO;
  if (diasFora >= limiarDias) return PERMANENCIA_LONGA;
  if (diasFora > prazoDias) return ATRASADO;
  return NO_PRAZO;
}

inline const char* nomeDoEstado(EstadoCiclo e) {
  switch (e) {
    case NO_PRAZO: return "NO_PRAZO";
    case ATRASADO: return "ATRASADO";
    case PERMANENCIA_LONGA: return "PERMANENCIA_LONGA";
    default: return "RETORNADO";
  }
}

// ---------- Debounce com o relógio injetado ----------
// O sketch passa millis(); o teste passa o instante que quiser.
struct Debounce {
  int estavel;          // última leitura aceita
  int anterior;         // última leitura vista, talvez um quique
  uint32_t mudouEmMs;   // instante da última mudança vista
};

inline Debounce novoDebounce(int leitura, uint32_t agoraMs) {
  Debounce d = { leitura, leitura, agoraMs };
  return d;
}

// Devolve true quando aceita uma mudança: o contato ficou parado pela janela inteira.
inline bool atualizarDebounce(Debounce& d, int leitura, uint32_t agoraMs, uint32_t janelaMs) {
  if (leitura != d.anterior) {
    d.anterior = leitura;
    d.mudouEmMs = agoraMs;
  }
  if (agoraMs - d.mudouEmMs >= janelaMs && leitura != d.estavel) {
    d.estavel = leitura;
    return true;
  }
  return false;
}

// ---------- Fila de eventos para envio posterior ----------
enum TipoEvento : uint8_t {
  EV_CADASTRO = 0, EV_LEITURA = 1, EV_LACRE = 2,
  EV_ALERTA_ATRASO = 3, EV_ALERTA_PERMANENCIA = 4, EV_RETORNO = 5
};

inline const char* nomeDoEvento(uint8_t tipo) {
  switch (tipo) {
    case EV_CADASTRO: return "cadastro";
    case EV_LEITURA: return "leitura";
    case EV_LACRE: return "lacre_rompido";
    case EV_ALERTA_ATRASO: return "alerta_atraso";
    case EV_ALERTA_PERMANENCIA: return "alerta_permanencia";
    default: return "retorno";
  }
}

struct Evento {
  uint16_t dia;
  uint8_t tipo;
  uint8_t bateriaPct;
};

const uint8_t CAPACIDADE_FILA = 8;

// Buffer circular da Aula 04. Cheia, a fila descarta o evento mais antigo e
// conta a perda, para o painel mostrar que houve descarte.
struct FilaEventos {
  Evento itens[CAPACIDADE_FILA];
  uint8_t inicio;
  uint8_t quantidade;
  uint16_t perdidos;
};

inline void limparFila(FilaEventos& f) {
  f.inicio = 0;
  f.quantidade = 0;
  f.perdidos = 0;
}

inline void guardarEvento(FilaEventos& f, Evento e) {
  if (f.quantidade == CAPACIDADE_FILA) {
    f.inicio = (f.inicio + 1) % CAPACIDADE_FILA;
    f.quantidade--;
    f.perdidos++;
  }
  f.itens[(f.inicio + f.quantidade) % CAPACIDADE_FILA] = e;
  f.quantidade++;
}

// Copia o evento mais antigo sem tirá-lo da fila: só sai depois de publicado.
inline bool primeiroEvento(const FilaEventos& f, Evento& saida) {
  if (f.quantidade == 0) return false;
  saida = f.itens[f.inicio];
  return true;
}

inline void descartarPrimeiro(FilaEventos& f) {
  if (f.quantidade == 0) return;
  f.inicio = (f.inicio + 1) % CAPACIDADE_FILA;
  f.quantidade--;
}

// Evento de alerta que a mudança de estado gera, ou -1 se não gera nenhum.
inline int eventoDaTransicao(EstadoCiclo antes, EstadoCiclo depois) {
  if (antes == depois) return -1;
  if (depois == ATRASADO) return EV_ALERTA_ATRASO;
  if (depois == PERMANENCIA_LONGA) return EV_ALERTA_PERMANENCIA;
  if (depois == RETORNADO) return EV_RETORNO;
  return -1;
}

// ---------- Bateria ----------
// Premissa simplificada para uma célula de íon-lítio: 4200 mV cheia, 3300 mV
// tratada como vazia, e reta entre as duas. A curva real não é reta.
const int BATERIA_CHEIA_MV = 4200;
const int BATERIA_VAZIA_MV = 3300;

inline int percentualBateria(int milivolts) {
  if (milivolts >= BATERIA_CHEIA_MV) return 100;
  if (milivolts <= BATERIA_VAZIA_MV) return 0;
  return (milivolts - BATERIA_VAZIA_MV) * 100 / (BATERIA_CHEIA_MV - BATERIA_VAZIA_MV);
}

// ---------- Autonomia estimada ----------
// Carga gasta numa semana, em mAh: despertares acordados mais o resto dormindo.
inline double cargaPorSemanaMah(double correnteAtivaMa, double segundosAcordado,
                                double correnteSonoMa, double intervaloSegundos) {
  const double semana = 7.0 * 24 * 3600;
  double despertares = semana / intervaloSegundos;
  double acordado = correnteAtivaMa * segundosAcordado * despertares;
  double dormindo = correnteSonoMa * (semana - segundosAcordado * despertares);
  return (acordado + dormindo) / 3600.0;
}

inline double autonomiaDias(double capacidadeMah, double correnteAtivaMa, double segundosAcordado,
                            double correnteSonoMa, double intervaloSegundos) {
  return capacidadeMah / cargaPorSemanaMah(correnteAtivaMa, segundosAcordado,
                                           correnteSonoMa, intervaloSegundos) * 7.0;
}

2. Contextualizar

Antes de rodar, escreva no PROCESSO.md:

  • o objetivo: o ciclo do equipamento EQ-02 (ciclo CI-009 do conjunto fictício, prazo de 30 dias) vai do cadastro ao retorno com um despertar por semana simulada;
  • o comportamento esperado em quatro pontos: com a rede desligada por três despertares, a fila chega a 3 eventos; ao religar, a fila vai a 0; no dia 35 aparece o alerta de atraso; no dia 154 aparece o de permanência longa;
  • o critério de aceitação: cada alerta é publicado uma única vez, e nenhum evento some sem aparecer na contagem de perdidos;
  • o que você pretende delegar à IA (os testes da etapa Confrontar) e o que faz sozinho.

Use o display e a serial para conferir os quatro pontos. Com DIAS_POR_DESPERTAR = 7, o dia 154 chega no 22º despertar. Para chegar mais depressa, troque por 30: os dias passam a ser 0, 30, 60 e assim por diante, o atraso aparece no dia 60 e a permanência no dia 150.

O LED de alerta acende em mostrar(), poucos milissegundos antes do sono, e o deep sleep desliga os pinos digitais: na placa, ele apaga enquanto o rastreador dorme. Para conferir o LED, use USAR_DEEP_SLEEP 0. Manter o LED aceso no sono com gpio_hold_en() custaria a corrente do próprio LED, ordens de grandeza acima dos 10 µA do chip.

3. Consultar

Cole no Gemini só o conteúdo de rastreador.h e peça: "Escreva testes em C++ para estas funções, sem biblioteca externa, com uma função main que rode todos." Registre o pedido exatamente como fez e salve a resposta num arquivo testes/testes_da_ia.cpp.

4. Confrontar

Um teste vale pelo defeito que ele pega. O jeito de saber é quebrar o código de propósito e ver quais testes continuam passando: esses não testam aquele ponto.

  1. Rode os testes da IA contra o rastreador.h original e anote quantos passam.
  2. Quebre o limiar: em estadoDoCiclo, troque diasFora >= limiarDias por diasFora > limiarDias. Com isso, o alerta do dia 150 some.
  3. Rode os testes da IA de novo. Os que ainda passam não testam a fronteira de 150 dias.
  4. Repita com outra quebra: apague f.perdidos++; em guardarEvento.
  5. Classifique cada teste da IA na tabela abaixo e desfaça as quebras.

Para rodar no computador, a partir da pasta do laboratório (no macOS, troque g++ por clang++; no Windows, use o g++ do WSL ou do MSYS2):

g++ -std=c++17 -Wall testes/testes_da_ia.cpp -o teste_ia && ./teste_ia
g++ -std=c++17 -Wall testes/teste_rastreador.cpp -o teste && ./teste

// Exemplo para a etapa Confrontar: testes no formato que assistentes de IA
// costumam devolver quando recebem o pedido "escreva testes para rastreador.h".
// Escritos para a aula como exemplo; nenhum veio de uma resposta real. Todos passam.
// A tarefa é descobrir quais deles continuariam passando com o código errado.
#include <cassert>
#include <cstdio>
#include "../rastreador.h"

// Teste 1: chama a função e imprime o resultado.
void testeEstadoFunciona() {
  EstadoCiclo e = estadoDoCiclo(160, 30, 150, false);
  std::printf("estado com 160 dias: %s\n", nomeDoEstado(e));
}

// Teste 2: compara a função com ela mesma.
void testeEstadoConsistente() {
  assert(estadoDoCiclo(100, 30, 150, false) == estadoDoCiclo(100, 30, 150, false));
}

// Teste 3: recalcula o esperado com a mesma regra da implementação,
// em passos de 20 dias que pulam o 150.
void testeEstadoCobreVariosDias() {
  for (int dias = 0; dias <= 300; dias += 20) {
    EstadoCiclo esperado = dias >= 150 ? PERMANENCIA_LONGA : (dias > 30 ? ATRASADO : NO_PRAZO);
    assert(estadoDoCiclo(dias, 30, 150, false) == esperado);
  }
}

// Teste 4: só valores longe das fronteiras.
void testePermanencia() {
  assert(estadoDoCiclo(10, 30, 150, false) == NO_PRAZO);
  assert(estadoDoCiclo(200, 30, 150, false) == PERMANENCIA_LONGA);
}

// Teste 5: confere só que nada estourou.
void testeFilaNaoQuebra() {
  FilaEventos f;
  limparFila(f);
  for (int i = 0; i < 20; i++) {
    Evento e = { (uint16_t)i, EV_LEITURA, 90 };
    guardarEvento(f, e);
  }
  assert(f.quantidade <= CAPACIDADE_FILA);
}

int main() {
  testeEstadoFunciona();
  testeEstadoConsistente();
  testeEstadoCobreVariosDias();
  testePermanencia();
  testeFilaNaoQuebra();
  std::printf("5 testes executados, todos passaram\n");
  return 0;
}

Suíte do laboratório (labs/aula10/testes/teste_rastreador.cpp) 

// Testes da lógica pura do rastreador, rodando no computador.
// Compilar e rodar a partir da pasta do laboratório:
//   g++ -std=c++17 -Wall testes/teste_rastreador.cpp -o teste && ./teste
// No macOS, troque g++ por clang++. Saída diferente de zero indica falha.
#include <cmath>
#include <cstdio>
#include "../rastreador.h"

static int verificacoes = 0;
static int falhas = 0;

#define VERIFICA(condicao)                                                   \
  do {                                                                       \
    verificacoes++;                                                          \
    if (!(condicao)) {                                                       \
      falhas++;                                                              \
      std::printf("FALHOU na linha %d: %s\n", __LINE__, #condicao);          \
    }                                                                        \
  } while (0)

static void testeEstadosNasFronteiras() {
  // Prazo de 30 dias e limiar de 150: os valores que importam são as bordas.
  VERIFICA(estadoDoCiclo(0, 30, 150, false) == NO_PRAZO);
  VERIFICA(estadoDoCiclo(30, 30, 150, false) == NO_PRAZO);      // no dia do prazo ainda está no prazo
  VERIFICA(estadoDoCiclo(31, 30, 150, false) == ATRASADO);
  VERIFICA(estadoDoCiclo(149, 30, 150, false) == ATRASADO);
  VERIFICA(estadoDoCiclo(150, 30, 150, false) == PERMANENCIA_LONGA);  // o TAPI alerta a partir de 150
  VERIFICA(estadoDoCiclo(190, 30, 150, false) == PERMANENCIA_LONGA);
  VERIFICA(estadoDoCiclo(190, 30, 150, true) == RETORNADO);      // retorno encerra qualquer estado
  VERIFICA(estadoDoCiclo(60, 30, 60, false) == PERMANENCIA_LONGA);  // limiar configurável
}

static void testeTransicoesGeramUmAlertaSo() {
  VERIFICA(eventoDaTransicao(NO_PRAZO, ATRASADO) == EV_ALERTA_ATRASO);
  VERIFICA(eventoDaTransicao(ATRASADO, PERMANENCIA_LONGA) == EV_ALERTA_PERMANENCIA);
  VERIFICA(eventoDaTransicao(ATRASADO, ATRASADO) == -1);         // sem repetir o alerta toda semana
  VERIFICA(eventoDaTransicao(ATRASADO, RETORNADO) == EV_RETORNO);
}

static void testeDebounceAceitaUmaMudancaPorRompimento() {
  // Lacre intacto lê LOW; rompido lê HIGH. Quiques em 0, 1, 3, 4 e 6 ms.
  Debounce d = novoDebounce(0, 0);
  int mudancas = 0;
  const int quiques[] = { 1, 0, 1, 0, 1 };
  const uint32_t instantes[] = { 0, 1, 3, 4, 6 };
  for (int i = 0; i < 5; i++) mudancas += atualizarDebounce(d, quiques[i], instantes[i], 20);
  VERIFICA(mudancas == 0);                                       // ainda quicando
  mudancas += atualizarDebounce(d, 1, 25, 20);
  VERIFICA(mudancas == 0);                                       // 19 ms parado: falta 1 ms
  mudancas += atualizarDebounce(d, 1, 26, 20);
  VERIFICA(mudancas == 1);
  VERIFICA(d.estavel == 1);
  mudancas += atualizarDebounce(d, 1, 500, 20);
  VERIFICA(mudancas == 1);                                       // lacre continua rompido: nada novo
}

static void testeFilaGuardaNaOrdemEDescartaOMaisAntigo() {
  FilaEventos f;
  limparFila(f);
  Evento e;
  VERIFICA(!primeiroEvento(f, e));
  for (int semana = 1; semana <= 10; semana++) {
    Evento novo = { (uint16_t)(semana * 7), EV_LEITURA, 90 };
    guardarEvento(f, novo);
  }
  VERIFICA(f.quantidade == CAPACIDADE_FILA);
  VERIFICA(f.perdidos == 2);                                     // 10 eventos numa fila de 8
  VERIFICA(primeiroEvento(f, e) && e.dia == 21);                 // os dias 7 e 14 foram descartados
  descartarPrimeiro(f);
  VERIFICA(primeiroEvento(f, e) && e.dia == 28);
  VERIFICA(f.quantidade == CAPACIDADE_FILA - 1);
}

static void testeBateria() {
  VERIFICA(percentualBateria(4200) == 100);
  VERIFICA(percentualBateria(3750) == 50);
  VERIFICA(percentualBateria(3300) == 0);
  VERIFICA(percentualBateria(5000) == 100);
  VERIFICA(percentualBateria(2000) == 0);
}

static void testeAutonomiaComAsPremissasDaAula() {
  // 100 mA acordado por 10 s, 10 µA dormindo, bateria de exemplo de 2000 mAh.
  const double semana = 7.0 * 24 * 3600;
  VERIFICA(std::fabs(cargaPorSemanaMah(100, 10, 0.010, semana) - 1.9578) < 0.001);
  VERIFICA(std::lround(autonomiaDias(2000, 100, 10, 0.010, semana)) == 7151);
  VERIFICA(std::lround(autonomiaDias(2000, 100, 10, 0.010, 3600)) == 290);
  VERIFICA(std::lround(autonomiaDias(2000, 100, 10, 1.0, semana)) == 83);
}

int main() {
  testeEstadosNasFronteiras();
  testeTransicoesGeramUmAlertaSo();
  testeDebounceAceitaUmaMudancaPorRompimento();
  testeFilaGuardaNaOrdemEDescartaOMaisAntigo();
  testeBateria();
  testeAutonomiaComAsPremissasDaAula();
  std::printf("%d verificações, %d falha(s)\n", verificacoes, falhas);
  return falhas == 0 ? 0 : 1;
}

  2. Partindo de um ciclo recém-cadastrado e publicado, a rede fica desligada por dez despertares e o lacre é rompido no primeiro. O evento do lacre chega ao broker? Que mudança em guardarEvento() daria prioridade aos alertas sobre as leituras de rotina?

7. Desafio: autonomia com as suas medidas

Documente a estimativa de autonomia do protótipo físico com as medidas e as premissas, numa tabela com a fonte de cada número:

  1. Na bancada, com o multímetro em série com a alimentação (autoestudo "Como usar o Multímetro", Semana 01), meça a corrente acordado na escala de mA, com USAR_DEEP_SLEEP 0 durante a publicação.
  2. Para a corrente dormindo, ligue o multímetro na escala de µA com um fio em paralelo com ele e retire o fio só depois que a placa dormir. Trocar a escala com o circuito ligado interrompe a alimentação, e o pico do Wi-Fi na escala de µA pode derrubar a placa ou queimar o fusível.
  3. Meça o tempo acordado com millis() e some o tempo de boot, que o millis() não conta.
  4. Declare a bateria que o grupo pretende usar e a capacidade do datasheet dela.
  5. Calcule a carga por semana e a autonomia com autonomiaDias() do rastreador.h e compare com a estimativa feita só com o datasheet do ESP32.
  6. Escreva, sem exagero, o que o cálculo deixa de fora: autodescarga da bateria, temperatura, regulador e conversor USB-serial da placa, tentativas de reconexão sem rede.
