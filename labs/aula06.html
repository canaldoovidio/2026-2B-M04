Laboratório: o rastreador publica o evento do ciclo em JSON por MQTT

O TAPI da Cerradinho pede que o rastreador informe status uma vez por semana e que o evento do lacre saia fora da rotina quando houver conexão. Aqui o ESP32 entra no Wi-Fi do simulador, conecta a um broker MQTT público e publica um JSON a cada 10 s (a semana acelerada) e a cada violação do lacre. Você assina os tópicos num cliente MQTT web e confere cada mensagem. O laboratório segue as cinco etapas do 5C no seu PROCESSO.md.

1. Abrir o projeto no Wokwi

  1. Abra wokwi.com/projects/new/esp32.
  2. Na aba diagram.json, apague tudo e cole o circuito abaixo: lacre no GPIO 4, LED de status no GPIO 2 e um potenciômetro no GPIO 34 fazendo o papel da bateria.
  3. Na aba Library Manager, clique no + e adicione PubSubClient. O Wokwi grava a lista no arquivo libraries.txt, que deve ficar igual ao bloco abaixo.
  4. Na aba sketch.ino, apague tudo, cole o código e troque t24d00 pelo identificador da sua dupla (por exemplo, t24d07). O broker é público e compartilhado: sem essa troca, as duplas leem as mensagens umas das outras.
  5. Inicie a simulação e abra o monitor serial. O Wokwi oferece a rede aberta Wokwi-GUEST no canal 6 e dá acesso de saída à internet.
  6. Abra o cliente MQTT web do HiveMQ em hivemq.com/demos/websocket-client, conecte em broker.hivemq.com com a porta de websocket que o cliente já traz e assine cerradinho/t24d00/#, trocando pelo identificador da sua dupla.

Código (labs/aula06/sketch.ino) 

// Aula 06: o rastreador da Cerradinho publica o evento do ciclo em JSON por MQTT.
// Wi-Fi do Wokwi (Wokwi-GUEST, canal 6) e broker público broker.hivemq.com,
// porta 1883 (P7: o broker de sala ainda não está definido). O broker público é
// compartilhado: só dados fictícios, nunca dado real da Cerradinho.
// Tópicos: cerradinho/<dupla>/dv01/status, .../lacre e .../comando.
// Troque DUPLA pelo número da sua dupla para não misturar com as outras.
#include <WiFi.h>
#include <PubSubClient.h>

const char* DUPLA = "t24d00";
const char* DISPOSITIVO = "dv01";
const char* BROKER = "broker.hivemq.com";
const int PORTA = 1883;

const int PINO_LACRE = 4;          // botão até o GND, com pull-up interno
const int PINO_LED = 2;            // LED de status
const int PINO_BATERIA = 34;       // potenciômetro no lugar do divisor da Aula 05
const unsigned long JANELA_DEBOUNCE_MS = 20;
const unsigned long INTERVALO_STATUS_MS = 10000;  // a "semana" do TAPI, acelerada para 10 s

// Ciclo aberto fictício de dados/ciclos.csv: EQ-01 está com o DV-01 desde
// 25/03/2026, 190 dias fora na data de referência 01/10/2026.
const char* CICLO = "CI-003";
const char* EQUIPAMENTO = "EQ-01";
const int DIAS_FORA = 190;

WiFiClient rede;
PubSubClient mqtt(rede);

char topico[48];
char payload[160];
unsigned long sequencia = 0;
unsigned long ultimoStatusMs = 0;
unsigned long ultimaTentativaMs = 0;
unsigned long naoEnviados = 0;
int leituraAnterior = HIGH;
int estadoEstavel = HIGH;
unsigned long ultimaMudancaMs = 0;

int bateriaPct() {
  // Simplificação: o potenciômetro vai de 0 a 4095 no ADC de 12 bits.
  return map(analogRead(PINO_BATERIA), 0, 4095, 0, 100);
}

bool publicarEvento(const char* tipo, const char* evento) {
  snprintf(topico, sizeof(topico), "cerradinho/%s/%s/%s", DUPLA, DISPOSITIVO, tipo);
  int n = snprintf(payload, sizeof(payload),
    "{\"dispositivo\":\"%s\",\"ciclo\":\"%s\",\"equipamento\":\"%s\","
    "\"evento\":\"%s\",\"dias_fora\":%d,\"bateria_pct\":%d,\"seq\":%lu}",
    DISPOSITIVO, CICLO, EQUIPAMENTO, evento, DIAS_FORA, bateriaPct(), ++sequencia);
  if (n >= (int)sizeof(payload)) {   // snprintf cortou o JSON: não publicar pela metade
    Serial.println("JSON maior que o buffer; evento não publicado");
    return false;
  }
  if (!mqtt.connected() || !mqtt.publish(topico, payload)) {
    naoEnviados++;                   // o desafio guarda e reenvia (buffer da Aula 04)
    Serial.print("Sem conexão; eventos não enviados: ");
    Serial.println(naoEnviados);
    return false;
  }
  Serial.print(topico);
  Serial.print(" <- ");
  Serial.println(payload);
  return true;
}

void aoReceber(char* topicoRecebido, byte* dados, unsigned int tamanho) {
  // Comando "piscar": ajuda a achar o equipamento certo dentro da oficina.
  if (tamanho == 6 && strncmp((const char*)dados, "piscar", 6) == 0) {
    for (int i = 0; i < 5; i++) {
      digitalWrite(PINO_LED, HIGH); delay(150);
      digitalWrite(PINO_LED, LOW);  delay(150);
    }
    publicarEvento("status", "localizar");
  }
}

void conectarMqtt() {
  if (mqtt.connected() || millis() - ultimaTentativaMs < 5000) return;
  ultimaTentativaMs = millis();      // reconexão sem travar o loop
  String id = String("m04-") + DUPLA + "-" + String((uint32_t)ESP.getEfuseMac(), HEX);
  if (mqtt.connect(id.c_str())) {
    snprintf(topico, sizeof(topico), "cerradinho/%s/%s/comando", DUPLA, DISPOSITIVO);
    mqtt.subscribe(topico);
    Serial.println("MQTT conectado ao broker");
    publicarEvento("status", "conectado");
  } else {
    Serial.print("MQTT falhou, estado ");
    Serial.println(mqtt.state());
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(PINO_LACRE, INPUT_PULLUP);
  pinMode(PINO_LED, OUTPUT);
  WiFi.begin("Wokwi-GUEST", "", 6);
  Serial.print("Conectando ao Wi-Fi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(250);
    Serial.print(".");
  }
  Serial.println(" conectado");
  mqtt.setServer(BROKER, PORTA);
  mqtt.setCallback(aoReceber);
}

void loop() {
  conectarMqtt();
  mqtt.loop();

  int leitura = digitalRead(PINO_LACRE);   // debounce da Aula 02
  if (leitura != leituraAnterior) {
    ultimaMudancaMs = millis();
    leituraAnterior = leitura;
  }
  if (millis() - ultimaMudancaMs >= JANELA_DEBOUNCE_MS && leitura != estadoEstavel) {
    estadoEstavel = leitura;
    if (estadoEstavel == LOW) publicarEvento("lacre", "violacao");  // fora da rotina, como o TAPI pede
  }

  if (millis() - ultimoStatusMs >= INTERVALO_STATUS_MS) {
    ultimoStatusMs = millis();
    publicarEvento("status", "rotina");
  }
}

O broker broker.hivemq.com, porta 1883, é o broker público do HiveMQ: livre para aprender e testar, sem garantia de disponibilidade e aberto a qualquer pessoa. Nada de dado real da Cerradinho passa por ele; o ciclo CI-003 e o equipamento EQ-01 são do conjunto fictício de dados/. O broker de sala ainda não foi definido (pendência P7 do plano de ensino). O Wokwi simula Wi-Fi e não simula LoRa nem LTE-M/NB-IoT: a comparação dessas tecnologias é feita no material de apoio e no deck.

2. Contextualizar

Antes de rodar, escreva no PROCESSO.md:

  • o objetivo: cada violação do lacre chega ao assinante como uma mensagem JSON no tópico .../lacre, e a rotina de status chega a cada 10 s em .../status;
  • o comportamento esperado: quantas mensagens o cliente web mostra em um minuto sem ninguém tocar no lacre, e quantas aparecem depois de três apertos;
  • o critério de aceitação: por exemplo, três apertos geram exatamente três mensagens de violação, com seq crescente e sem lacuna;
  • a decisão sobre o tópico: o TAPI pede um dispositivo reutilizável, que passa de um equipamento para outro. O tópico identifica o dispositivo ou o equipamento? Escreva a escolha da dupla e o motivo;
  • o que você pretende delegar à IA e o que pretende fazer sozinho.

6. Comprovar

Sem consultar a IA, responda no PROCESSO.md:

  1. Por que uma assinatura em cerradinho/t24d00/+/lacre recebe a violação do dv01 e não receberia um tópico com um nível a mais, como .../dv01/lacre/autorizada?
  2. Troque a constante BROKER por "broker.invalido.exemplo", rode e aperte o lacre três vezes. O que acontece com essas violações, e qual linha do sketch decide isso? Depois volte ao broker original.
  3. Publique a mensagem piscar no tópico .../dv01/comando pelo cliente web. O que o rastreador faz, e por que isso ajuda a achar o equipamento numa oficina?

7. Desafio: reenviar o que ficou sem conexão

O TAPI pede armazenamento de eventos e envio posterior quando a conexão falta. Junte o buffer circular da Aula 04 a este sketch: quando publicarEvento falhar, guarde o JSON no buffer em vez de só contar; quando conectarMqtt reconectar, publique os eventos guardados na ordem em que aconteceram. Para simular a falta de cobertura, acrescente uma chave deslizante (wokwi-slide-switch) que, ligada, chama mqtt.disconnect() e impede a reconexão. Comprove no cliente web que os seq chegam completos e em ordem.
